class DeploymentManager:
    def __init__(
        self,
        namespace: str = "ai-system",
        replicas: int = 3,
        gpu_enabled: bool = True
    ):
        self.namespace = namespace
        self.replicas = replicas
        self.gpu_enabled = gpu_enabled
        
        # Initialize Kubernetes client
        config.load_kube_config()
        self.k8s_client = client.ApiClient()
        self.k8s_apps = client.AppsV1Api()
        self.k8s_core = client.CoreV1Api()
        
        # Initialize Docker client
        self.docker_client = docker.from_env()
        
        # Initialize metrics
        self.request_counter = Counter(
            'ai_system_requests_total',
            'Total number of requests processed'
        )
        self.processing_time = Histogram(
            'ai_system_processing_seconds',
            'Time spent processing requests'
        )
        self.gpu_utilization = Gauge(
            'ai_system_gpu_utilization',
            'GPU utilization percentage'
        )
        
        # Initialize Ray
        ray.init(address='auto', namespace=namespace)
        
    def deploy(self):
        self._create_namespace()
        self._deploy_monitoring()
        self._deploy_security()
        self._deploy_model_servers()
        self._deploy_api_gateway()
        self._setup_autoscaling()
        
    def _create_namespace(self):
        try:
            namespace = client.V1Namespace(
                metadata=client.V1ObjectMeta(name=self.namespace)
            )
            self.k8s_core.create_namespace(namespace)
        except ApiException as e:
            if e.status != 409:  # Ignore if namespace already exists
                raise
                
    def _deploy_monitoring(self):
        # Deploy Prometheus
        prometheus_deployment = {
            'apiVersion': 'apps/v1',
            'kind': 'Deployment',
            'metadata': {
                'name': 'prometheus',
                'namespace': self.namespace
            },
            'spec': {
                'replicas': 1,
                'selector': {
                    'matchLabels': {
                        'app': 'prometheus'
                    }
                },
                'template': {
                    'metadata': {
                        'labels': {
                            'app': 'prometheus'
                        }
                    },
                    'spec': {
                        'containers': [{
                            'name': 'prometheus',
                            'image': 'prom/prometheus:latest',
                            'ports': [{
                                'containerPort': 9090
                            }],
                            'volumeMounts': [{
                                'name': 'prometheus-config',
                                'mountPath': '/etc/prometheus'
                            }]
                        }],
                        'volumes': [{
                            'name': 'prometheus-config',
                            'configMap': {
                                'name': 'prometheus-config'
                            }
                        }]
                    }
                }
            }
        }
        
        # Deploy Grafana
        grafana_deployment = {
            'apiVersion': 'apps/v1',
            'kind': 'Deployment',
            'metadata': {
                'name': 'grafana',
                'namespace': self.namespace
            },
            'spec': {
                'replicas': 1,
                'selector': {
                    'matchLabels': {
                        'app': 'grafana'
                    }
                },
                'template': {
                    'metadata': {
                        'labels': {
                            'app': 'grafana'
                        }
                    },
                    'spec': {
                        'containers': [{
                            'name': 'grafana',
                            'image': 'grafana/grafana:latest',
                            'ports': [{
                                'containerPort': 3000
                            }]
                        }]
                    }
                }
            }
        }
        
        try:
            self.k8s_apps.create_namespaced_deployment(
                namespace=self.namespace,
                body=prometheus_deployment
            )
            self.k8s_apps.create_namespaced_deployment(
                namespace=self.namespace,
                body=grafana_deployment
            )
        except ApiException as e:
            if e.status != 409:
                raise
