from __future__ import annotations
import numpy as np
from typing import Dict, List, Optional, Set, Tuple
from dataclasses import dataclass, field
import asyncio
import logging
from datetime import datetime
from concurrent.futures import ThreadPoolExecutor

from ..memory.graph import MemoryGraph
from ..memory.bank import MemoryBank
from ..core.nodes.pure_node import PureNode

logger = logging.getLogger(__name__)

@dataclass
class MemoryMetrics:
    usage: float = 0.0
    fragmentation: float = 0.0
    access_patterns: Dict[str, int] = field(default_factory=dict)
    cache_hits: int = 0
    cache_misses: int = 0
    optimization_count: int = 0

class MemoryOptimizer:
    def __init__(
        self,
        memory_graph: MemoryGraph,
        memory_bank: MemoryBank,
        cleanup_threshold: float = 0.8,
        optimization_interval: int = 1000
    ):
        self.memory_graph = memory_graph
        self.memory_bank = memory_bank
        self.cleanup_threshold = cleanup_threshold
        self.optimization_interval = optimization_interval
        self.metrics = MemoryMetrics()
        
        # Caching
        self.cache: Dict[str, Any] = {}
        self.cache_priority: Dict[str, float] = {}
        self.max_cache_size = 10000
        
        # Access tracking
        self.access_history: List[Tuple[str, datetime]] = []
        self.frequently_accessed: Set[str] = set()
        
        self.thread_pool = ThreadPoolExecutor(max_workers=4)
        logger.info("Memory optimizer initialized")

    async def optimize(self):
        """Performs memory optimization."""
        try:
            if self._should_optimize():
                # Analyze current state
                memory_state = self._analyze_memory_state()
                
                # Perform optimizations
                if memory_state["fragmentation"] > 0.3:
                    await self._defragment_memory()
                    
                if memory_state["usage"] > self.cleanup_threshold:
                    await self._cleanup_memory()
                    
                await self._optimize_cache(memory_state)
                await self._reorganize_pathways()
                
                self.metrics.optimization_count += 1
                
        except Exception as e:
            logger.error(f"Memory optimization error: {str(e)}")

    def _analyze_memory_state(self) -> Dict[str, float]:
        """Analyzes current memory state."""
        try:
            usage = len(self.memory_graph.nodes) / self.memory_graph.capacity
            
            # Calculate fragmentation
            fragmentation = self._calculate_fragmentation()
            
            # Analyze access patterns
            access_patterns = self._analyze_access_patterns()
            
            # Calculate cache efficiency
            cache_efficiency = (
                self.metrics.cache_hits /
                max(1, self.metrics.cache_hits + self.metrics.cache_misses)
            )
            
            return {
                "usage": usage,
                "fragmentation": fragmentation,
                "access_patterns": access_patterns,
                "cache_efficiency": cache_efficiency
            }
            
        except Exception as e:
            logger.error(f"Memory state analysis error: {str(e)}")
            return {
                "usage": 0.0,
                "fragmentation": 0.0,
                "access_patterns": {},
                "cache_efficiency": 0.0
            }

    async def _defragment_memory(self):
        """Defragments memory by reorganizing nodes and connections."""
        try:
            # Group related nodes
            node_clusters = self._identify_node_clusters()
            
            # Reorganize nodes within clusters
            for cluster in node_clusters:
                await self._optimize_cluster(cluster)
                
            # Update pathways
            await self._update_pathways(node_clusters)
            
            logger.info(f"Memory defragmentation complete: {len(node_clusters)} clusters")
            
        except Exception as e:
            logger.error(f"Memory defragmentation error: {str(e)}")

    async def _cleanup_memory(self):
        """Removes unused or low-priority memory entries."""
        try:
            # Calculate node scores
            node_scores = {}
            for node_id, node in self.memory_graph.nodes.items():
                score = self._calculate_node_score(node)
                node_scores[node_id] = score
                
            # Identify nodes for removal
            threshold = np.percentile(list(node_scores.values()), 20)
            nodes_to_remove = [
                node_id for node_id, score in node_scores.items()
                if score < threshold and node_id not in self.frequently_accessed
            ]
            
            # Remove nodes
            for node_id in nodes_to_remove:
                await self._remove_node(node_id)
                
            logger.info(f"Memory cleanup complete: removed {len(nodes_to_remove)} nodes")
            
        except Exception as e:
            logger.error(f"Memory cleanup error: {str(e)}")

    async def _optimize_cache(self, memory_state: Dict[str, float]):
        """Optimizes memory cache based on usage patterns."""
        try:
            # Update cache priorities
            current_time = datetime.now()
            for key in list(self.cache.keys()):
                # Calculate priority based on access frequency and recency
                access_count = self.metrics.access_patterns.get(key, 0)
                last_access = next(
                    (time for id, time in reversed(self.access_history)
                     if id == key),
                    None
                )
                
                if last_access:
                    time_factor = 1.0 / (
                        1.0 + (current_time - last_access).total_seconds() / 3600
                    )
                    priority = access_count * time_factor
                    self.cache_priority[key] = priority
                    
            # Remove low priority items if cache is full
            if len(self.cache) > self.max_cache_size:
                await self._trim_cache()
                
            # Prefetch frequently accessed items
            await self._prefetch_frequent_items()
            
        except Exception as e:
            logger.error(f"Cache optimization error: {str(e)}")

    async def _reorganize_pathways(self):
        """Reorganizes memory pathways for optimal access."""
        try:
            # Analyze pathway usage
            pathway_usage = self._analyze_pathway_usage()
            
            # Optimize high-traffic pathways
            for pathway_id, usage in pathway_usage.items():
                if usage > 100:  # High traffic threshold
                    await self._optimize_pathway(pathway_id)
                    
            # Remove unused pathways
            await self._cleanup_pathways()
            
            logger.info("Pathway reorganization complete")
            
        except Exception as e:
            logger.error(f"Pathway reorganization error: {str(e)}")

    def _calculate_node_score(self, node: PureNode) -> float:
        """Calculates importance score for a node."""
        try:
            # Base score from node properties
            score = node.state.health * node.dna.traits.get("importance", 0.5)
            
            # Access frequency factor
            access_count = self.metrics.access_patterns.get(node.node_id, 0)
            access_factor = min(1.0, access_count / 100)  # Cap at 100 accesses
            
            # Connection importance
            connection_count = len(node.state.connections)
            connection_factor = min(1.0, connection_count / 10)  # Cap at 10 connections
            
            # Combine factors
            final_score = (
                0.4 * score +
                0.3 * access_factor +
                0.3 * connection_factor
            )
            
            return final_score
            
        except Exception as e:
            logger.error(f"Node score calculation error: {str(e)}")
            return 0.0

    def _calculate_fragmentation(self) -> float:
        """Calculates memory fragmentation level."""
        try:
            if not self.memory_graph.nodes:
                return 0.0
                
            # Calculate average connection distance
            distances = []
            for node_id, node in self.memory_graph.nodes.items():
                for conn_id in node.state.connections:
                    if conn_id in self.memory_graph.nodes:
                        distance = self._calculate_node_distance(
                            node_id,
                            conn_id
                        )
                        distances.append(distance)
                        
            if not distances:
                return 0.0
                
            # Normalize distances
            max_distance = max(distances)
            if max_distance == 0:
                return 0.0
                
            normalized_distances = [d / max_distance for d in distances]
            
            # Calculate fragmentation score
            return np.mean(normalized_distances)
            
        except Exception as e:
            logger.error(f"Fragmentation calculation error: {str(e)}")
            return 0.0

    def _analyze_access_patterns(self) -> Dict[str, int]:
        """Analyzes memory access patterns."""
        try:
            current_time = datetime.now()
            recent_window = 3600  # 1 hour
            
            # Count recent accesses
            access_counts = {}
            for node_id, access_time in self.access_history:
                if (current_time - access_time).total_seconds() <= recent_window:
                    access_counts[node_id] = access_counts.get(node_id, 0) + 1
                    
            return access_counts
            
        except Exception as e:
            logger.error(f"Access pattern analysis error: {str(e)}")
            return {}

    async def _trim_cache(self):
        """Removes low priority items from cache."""
        try:
            # Sort items by priority
            sorted_items = sorted(
                self.cache_priority.items(),
                key=lambda x: x[1]
            )
            
            # Remove bottom 20%
            items_to_remove = sorted_items[:len(sorted_items) // 5]
            for key, _ in items_to_remove:
                del self.cache[key]
                del self.cache_priority[key]
                
        except Exception as e:
            logger.error(f"Cache trimming error: {str(e)}")

    async def _prefetch_frequent_items(self):
        """Prefetches frequently accessed items into cache."""
        try:
            for node_id, access_count in self.metrics.access_patterns.items():
                if (
                    access_count > 10 and  # Frequently accessed
                    node_id not in self.cache and  # Not in cache
                    len(self.cache) < self.max_cache_size  # Cache has space
                ):
                    node = self.memory_graph.get_node(node_id)
                    if node:
                        self.cache[node_id] = node
                        self.cache_priority[node_id] = access_count
                        
        except Exception as e:
            logger.error(f"Prefetch error: {str(e)}")

    async def _optimize_pathway(self, pathway_id: str):
        """Optimizes a specific memory pathway."""
        try:
            # Get pathway metrics
            metrics = self.memory_bank.get_pathway_metrics(pathway_id)
            
            if metrics["congestion"] > 0.8:
                # Create parallel pathway
                await self.memory_bank.create_parallel_pathway(pathway_id)
                
            elif metrics["error_rate"] > 0.1:
                # Reinforce pathway
                await self.memory_bank.reinforce_pathway(pathway_id)
                
        except Exception as e:
            logger.error(f"Pathway optimization error: {str(e)}")

    def _identify_node_clusters(self) -> List[Set[str]]:
        """Identifies clusters of related nodes."""
        try:
            clusters = []
            visited = set()
            
            for node_id in self.memory_graph.nodes:
                if node_id not in visited:
                    # Find connected component
                    cluster = self._find_connected_component(node_id)
                    clusters.append(cluster)
                    visited.update(cluster)
                    
            return clusters
            
        except Exception as e:
            logger.error(f"Cluster identification error: {str(e)}")
            return []

    def _find_connected_component(self, start_node: str) -> Set[str]:
        """Finds all nodes connected to start_node."""
        try:
            component = {start_node}
            queue = [start_node]
            
            while queue:
                node_id = queue.pop(0)
                node = self.memory_graph.get_node(node_id)
                
                if node:
                    for conn_id in node.state.connections:
                        if conn_id not in component:
                            component.add(conn_id)
                            queue.append(conn_id)
                            
            return component
            
        except Exception as e:
            logger.error(f"Connected component error: {str(e)}")
            return {start_node}

    async def _optimize_cluster(self, cluster: Set[str]):
        """Optimizes node organization within a cluster."""
        try:
            # Calculate node relationships
            relationships = {}
            for node_id in cluster:
                node = self.memory_graph.get_node(node_id)
                if node:
                    relationships[node_id] = {
                        conn_id: self._calculate_relationship_strength(
                            node_id,
                            conn_id
                        )
                        for conn_id in node.state.connections
                        if conn_id in cluster
                    }
                    
            # Reorganize based on relationships
            await self._reorganize_nodes(relationships)
            
        except Exception as e:
            logger.error(f"Cluster optimization error: {str(e)}")

    def _calculate_relationship_strength(
        self,
        node1_id: str,
        node2_id: str
    ) -> float:
        """Calculates strength of relationship between nodes."""
        try:
            node1 = self.memory_graph.get_node(node1_id)
            node2 = self.memory_graph.get_node(node2_id)
            
            if not node1 or not node2:
                return 0.0
                
            # Consider various factors
            factors = []
            
            # Connection frequency
            conn_count = sum(
                1 for conn in node1.state.connections
                if conn == node2_id
            )
            factors.append(min(1.0, conn_count / 10))
            
            # DNA similarity
            dna_similarity = self._calculate_dna_similarity(
                node1.dna,
                node2.dna
            )
            factors.append(dna_similarity)
            
            # Access pattern similarity
            access_similarity = self._calculate_access_similarity(
                node1_id,
                node2_id
            )
            factors.append(access_similarity)
            
            return np.mean(factors)
            
        except Exception as e:
            logger.error(f"Relationship strength calculation error: {str(e)}")
            return 0.0
